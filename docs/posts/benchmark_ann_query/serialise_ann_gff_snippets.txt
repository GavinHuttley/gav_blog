    ```python linenums="1"
    def bp(path: pathlib.Path, out: pathlib.Path) -> None:
        import pickle

        from BCBio import GFF

        with open(path) as in_handle:
            records = list(GFF.parse(in_handle))
        with out.open("wb") as f:
            pickle.dump(records, f)
    ```
